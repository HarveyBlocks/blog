# 生成随机密码

```java
import java.util.*;

/**
 * @author HarveyBlocks
 * @date 2023/09/22 12:28
 **/
public class CreatRandomPassword {
    private static final char[] CHARS ={
            '(' , ')' , '`' ,
            '!' , '@' , '#' , '$' ,
            '%' , '^' , '&' , '*' ,
            '_' , '-' , '+' ,
            '=' ,'|','{','}','[',']',
            ':' , ';' , '<' , '>' ,
            ',' , '.' , '?' ,
    } ;
    private List<Character> list;
    private int times;
    private int length;
    public CreatRandomPassword(int length,int times){
        this.times = length;
        this.length = times;
        this.list = new ArrayList<>();
    }
    public CreatRandomPassword(){
        this(16,1);
    }
    public void printPassword() {
        creatCharList();

        Scanner scanner = new Scanner(System.in);

        System.out.print("请输入你想要的密码长度:");
        if (scanner.hasNextInt()) length = scanner.nextInt();

        System.out.print("请输入你想要的密码的个数:");
        if (scanner.hasNextInt()) times = scanner.nextInt();

        scanner.close();

        creatPassword();
    }
    public void creatCharList(){
        for (char c: CHARS) {
            for (int j = 0; j < 3; j++) {
                list.add(c);
            }
        }
        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < 4; j++) {
                list.add((char) ('A' + i));
                list.add((char) ('a' + i));
            }
        }
        for (int i = 0; i < 10; i++) {
            for (int j = 0; j < 15; j++) {
                list.add((char) ('0' + i));
            }
        }
    }
    public void creatPassword(){
        Random random = new Random(Calendar.getInstance().getTimeInMillis());

        System.out.println("\n以下是为您生成的密码:");
        String password ="";
        for (int i = 0; i < times; i++) {
            password = "";
            for (int j = 0; j < length; j++) {
                password+=list.get(random.nextInt(list.size()));
            }
            System.out.println(password);
        }
    }
}
```

```java
public class Main {
    public static void main(String[] args) {
        new CreatRandomPassword().printPassword();
    }
}
```
