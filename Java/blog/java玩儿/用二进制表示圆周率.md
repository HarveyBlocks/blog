```java
public class Main {
    static final Double PI = 3.14159265358979323;
    public static void main(String[] args) {
        String str="11.";//二进制整数部分
        double pi=3;
        int i=0;//指向二进制小数点后一位
        while(pi!=PI){
            i += 1;
            if(pi+Math.pow(2,-i)<PI){
                str += "1";
                pi += Math.pow(2, -i);
            }else {
                str += "0";
            }
            System.out.println(str);
            System.out.println(pi);
            System.out.println(i);

            if(i>=55){
                break;
            }
        }

    }
}
```

用BigDecimal改进的2.0

```java
import java.math.BigDecimal;

public class Main {
    static final BigDecimal PI =new BigDecimal("3.14159265358979323") ;
    public static void main(String[] args) {
        String str="11.";//二进制整数部分
        BigDecimal pi =new BigDecimal("3.0");
        int i=0;//指向二进制小数点后一位
        while(!pi.equals(PI)){
            i += 1;
            if(pi.add(power(2,-i)).compareTo(PI)<0){
                str += "1";
                pi =pi.add(power(2, -i));
            }else {
                str += "0";
            }
            System.out.println(str);
            System.out.println(pi);
            System.out.println(i);

            if(i>=55){
                break;
            }
        }

    }
    public  static BigDecimal power(int a,int b){
        boolean flag = true;
        if (b<0){
            flag = false;
            b = -b;
        }else if (b==0&&a!=0) {
            return new BigDecimal("1.0");
        } else if(a==0){
            return new BigDecimal("0.0");//为考虑0的0次的情况，不过算了
        }
        BigDecimal ans = new BigDecimal("1.0");
        for (int i = 0; i < b; i++) {
            ans = ans.multiply(new BigDecimal(a+""));
        }
        if(!flag){
            ans=new BigDecimal("1.0").divide(ans,100,BigDecimal.ROUND_HALF_UP);
        }
        return ans;
    }
}
```
