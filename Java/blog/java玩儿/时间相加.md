import java.util.Scanner;
/**
 * 这个类的main()方法实现了将输入的时间相加并较整齐地输出
 */
public class sumTime {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int min = 0;
        int sec = 0;
        int hour = 0;
        System.out.println("按照mm-ss的形式输入时间,再回车,否则退出");
        while(true){
            System.out.print("min = ");
            if (!scanner.hasNextInt()) break;
            min += scanner.nextInt();
            System.out.print("sec = ");
            if (!scanner.hasNextInt()) break;
            sec += scanner.nextInt();
            System.out.println();
        }
        scanner.close();
        min = min + sec/60;
        hour = min/ 60;
        min %= 60;
        sec %= 60;
        System.out.print("时间共为:");
        System.out.println(
                timeFill(hour + "") + ":"
                        + timeFill(min + "") + ":"
                        + timeFill(sec + "")
        );

    }
    /**
     * timeFill()方法通过fill()方法实现对时间的格式化,
     * 但没这么智能,有瑕疵,导致它不再具有普适性
     */
    public static String timeFill(String oldStr){//时间 补充
        if (Integer.valueOf(oldStr)>99) return oldStr;//小时超过99
        return fill(oldStr, '0', 2);
    }
    /**
     * fill()方法用于填充一个字符串,完成后字符串将"后对齐"
     *oldStr:原来的想要填充的字符串
     *sign:用来填充的字符
     *lengthWant:填充后最终得到的长度
     */
    public static String fill(String oldStr,char sign,int lengthWant){
        if(oldStr.length()>lengthWant) return null;
        String newStr="";
        for (int i = 0; i < (lengthWant - oldStr.length()); i++) {
            newStr += sign;
        }
        return newStr + oldStr;
    }
}
